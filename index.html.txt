  <script>
    async function loadData() {
      try {
        const res = await fetch('data.json?t=' + Date.now());
        const data = await res.json();

        // 1. Render Header
        document.getElementById('last-updated').innerText = data.last_updated;

        // 2. Render Card 1 (Confidence & Market)
        const score = data.market.confidence_score;
        document.getElementById('confidence-score').innerText = score;
        document.getElementById('confidence-bar').style.width = score + '%';
        document.getElementById('spy-status').innerText = data.market.spy_status;
        document.getElementById('qqq-status').innerText = data.market.qqq_status;
        document.getElementById('vix-value').innerText = data.market.vix_val;

        const badge = document.getElementById('allocation-badge');
        const summary = document.getElementById('action-summary');
        const directive = document.getElementById('action-directive');
        const regimeStatus = document.getElementById('regime-status');

        if (score > 75) {
          badge.className = "px-2.5 py-1 text-xs font-bold rounded-full bg-emerald-500/20 text-emerald-400 border border-emerald-500/30";
          badge.innerText = "全倉執行 (>75)";
          regimeStatus.innerText = "牛市多頭輪動良好";
          summary.innerText = "市場結構極佳，各板塊輪動正常。建議執行全倉/高 Conviction 策略部署。";
          directive.className = "px-4 py-2 rounded-lg bg-emerald-600 text-white font-bold text-sm";
          directive.innerText = "建議持倉：100% 全倉";
        } else if (score >= 60) {
          badge.className = "px-2.5 py-1 text-xs font-bold rounded-full bg-amber-500/20 text-amber-400 border border-amber-500/30";
          badge.innerText = "選擇性部署 25-50% (60-75)";
          regimeStatus.innerText = "震盪中性 / 擇優進場";
          summary.innerText = "市場趨勢分化或波動上升。僅選擇最高 Conviction 標的，總倉位控制在 25-50% 之間。";
          directive.className = "px-4 py-2 rounded-lg bg-amber-600 text-white font-bold text-sm";
          directive.innerText = "建議持倉：25% - 50%";
        } else {
          badge.className = "px-2.5 py-1 text-xs font-bold rounded-full bg-rose-500/20 text-rose-400 border border-rose-500/30";
          badge.innerText = "空倉 / 防禦策略 (<60)";
          regimeStatus.innerText = "高風險 / 破位下跌";
          summary.innerText = "市場環境劣勢，VIX 或板塊破位指標觸發。停止新建立長頭寸，保持空倉或嚴格避險。";
          directive.className = "px-4 py-2 rounded-lg bg-rose-600 text-white font-bold text-sm";
          directive.innerText = "建議持倉：0% (空倉避險)";
        }

        // 3. Render Card 2 (Tonight Trades)
        document.getElementById('tonight-trades').innerHTML = data.tonight_trades.map(t => `
          <div class="flex justify-between items-center p-2.5 bg-gray-950/60 rounded border border-gray-800/80">
            <div>
              <span class="font-bold text-white mr-2">${t.ticker}</span>
              <span class="text-xs text-gray-400">${t.strategy}</span>
            </div>
            <div class="flex items-center gap-2">
              <span class="text-xs font-mono bg-blue-900/40 text-blue-300 px-2 py-0.5 rounded border border-blue-700/30">Conviction ${t.conviction}</span>
            </div>
          </div>
        `).join('');

        // 4. Render Card 3 (Long-term)
        document.getElementById('long-term-picks').innerHTML = data.long_term_picks.map(p => `
          <div class="flex justify-between items-center p-2.5 bg-gray-950/60 rounded border border-gray-800/80">
            <div>
              <span class="font-bold text-emerald-400 mr-2">${p.ticker}</span>
              <span class="text-xs text-gray-400">${p.reason}</span>
            </div>
            <span class="text-xs font-mono text-gray-300">${p.score}分</span>
          </div>
        `).join('');

        // 5. Render Card 4 (Avoid)
        document.getElementById('avoid-list').innerHTML = data.avoid_list.map(a => `
          <div class="flex justify-between items-center p-2.5 bg-gray-950/60 rounded border border-gray-800/80">
            <div>
              <span class="font-bold text-rose-400 mr-2">${a.ticker}</span>
              <span class="text-xs text-gray-400">${a.reason}</span>
            </div>
            <span class="text-xs font-mono text-rose-300/80 bg-rose-950/40 px-2 py-0.5 rounded">${a.risk_level}</span>
          </div>
        `).join('');

        lucide.createIcons();
      } catch (err) {
        console.error("載入數據失敗:", err);
      }
    }

    loadData();
  </script>
